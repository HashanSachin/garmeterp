Data Utilities & Security

1-Click Supabase SQL Schema

Copy and run this inside your Supabase project's SQL Editor to create all 4 factory tables

-- Run this inside Supabase SQL Editor (100% Free Tier Compatible)

CREATE TABLE IF NOT EXISTS employees (
  id TEXT PRIMARY KEY,
  emp_code TEXT NOT NULL,
  name TEXT NOT NULL,
  department TEXT NOT NULL,
  role TEXT DEFAULT 'Operator',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS attendance_logs (
  id TEXT PRIMARY KEY,
  employee_id TEXT NOT NULL,
  emp_code TEXT NOT NULL,
  employee_name TEXT NOT NULL,
  department TEXT NOT NULL,
  date TEXT NOT NULL,
  time_in TIMESTAMPTZ NOT NULL,
  time_out TIMESTAMPTZ,
  hours_worked NUMERIC DEFAULT 0,
  note TEXT DEFAULT '',
  status TEXT DEFAULT 'INSIDE'
);

CREATE TABLE IF NOT EXISTS stock_items (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  category TEXT NOT NULL,
  unit TEXT NOT NULL,
  min_stock NUMERIC DEFAULT 10,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS stock_movements (
  id TEXT PRIMARY KEY,
  item_id TEXT NOT NULL,
  item_name TEXT NOT NULL,
  category TEXT NOT NULL,
  unit TEXT NOT NULL,
  type TEXT NOT NULL, -- 'IN' or 'OUT'
  quantity NUMERIC NOT NULL,
  date TEXT NOT NULL,
  timestamp TIMESTAMPTZ DEFAULT NOW(),
  note TEXT DEFAULT ''
);

-- Enable public read/write policies for Factory Gate Kiosk & Manager App
ALTER TABLE employees ENABLE ROW LEVEL SECURITY;
ALTER TABLE attendance_logs ENABLE ROW LEVEL SECURITY;
ALTER TABLE stock_items ENABLE ROW LEVEL SECURITY;
ALTER TABLE stock_movements ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Public factory access employees" ON employees FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Public factory access attendance" ON attendance_logs FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Public factory access stock_items" ON stock_items FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Public factory access stock_movements" ON stock_movements FOR ALL USING (true) WITH CHECK (true);
